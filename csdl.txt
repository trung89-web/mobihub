-- Chay script nay trong MySQL truoc khi khoi dong ung dung.
CREATE DATABASE IF NOT EXISTS quan_ly_ban_dien_thoai
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE quan_ly_ban_dien_thoai;

CREATE TABLE IF NOT EXISTS danh_muc (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    ten_danh_muc VARCHAR(120) NOT NULL UNIQUE,
    mo_ta VARCHAR(500)
);

CREATE TABLE IF NOT EXISTS san_pham (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    ma_san_pham VARCHAR(40) NOT NULL UNIQUE,
    ten_san_pham VARCHAR(180) NOT NULL,
    thuong_hieu VARCHAR(100),
    danh_muc_id BIGINT NOT NULL,
    gia DECIMAL(15,2) NOT NULL,
    so_luong_ton INT NOT NULL DEFAULT 0,
    thoi_han_bao_hanh_thang INT NOT NULL DEFAULT 12,
    anh_url VARCHAR(1000),
    mo_ta TEXT,
    dang_kinh_doanh BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT chk_san_pham_gia CHECK (gia >= 0),
    CONSTRAINT chk_san_pham_ton CHECK (so_luong_ton >= 0),
    CONSTRAINT fk_san_pham_danh_muc FOREIGN KEY (danh_muc_id) REFERENCES danh_muc(id)
);

CREATE TABLE IF NOT EXISTS khach_hang (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    ho_ten VARCHAR(150) NOT NULL,
    so_dien_thoai VARCHAR(20) NOT NULL UNIQUE,
    email VARCHAR(150),
    dia_chi VARCHAR(500),
    ngay_sinh DATE
);

    'ALTER TABLE san_pham ADD COLUMN thoi_han_bao_hanh_thang INT NOT NULL DEFAULT 12', 'SELECT 1');
PREPARE mobihub_stmt FROM @mobihub_ddl;
EXECUTE mobihub_stmt;
DEALLOCATE PREPARE mobihub_stmt;

CREATE TABLE IF NOT EXISTS dia_chi_khach_hang (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    khach_hang_id BIGINT NOT NULL,
    nhan VARCHAR(60) NOT NULL DEFAULT 'Nhà riêng',
    nguoi_nhan VARCHAR(150) NOT NULL,
    so_dien_thoai VARCHAR(20) NOT NULL,
    dia_chi VARCHAR(500) NOT NULL,
    mac_dinh BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT fk_dia_chi_khach_hang FOREIGN KEY (khach_hang_id)
        REFERENCES khach_hang(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS tai_khoan (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    khach_hang_id BIGINT,
    email VARCHAR(150) NOT NULL UNIQUE,
    mat_khau_hash VARCHAR(100) NOT NULL,
    vai_tro VARCHAR(20) NOT NULL,
    phuong_thuc_thanh_toan_mac_dinh VARCHAR(40) NOT NULL DEFAULT 'COD',
    dang_hoat_dong BOOLEAN NOT NULL DEFAULT TRUE,
    ngay_tao DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_tai_khoan_khach_hang FOREIGN KEY (khach_hang_id)
        REFERENCES khach_hang(id) ON DELETE CASCADE
);

    'ALTER TABLE tai_khoan ADD COLUMN phuong_thuc_thanh_toan_mac_dinh VARCHAR(40) NOT NULL DEFAULT ''COD''', 'SELECT 1');
PREPARE mobihub_stmt FROM @mobihub_ddl;
EXECUTE mobihub_stmt;
DEALLOCATE PREPARE mobihub_stmt;

CREATE TABLE IF NOT EXISTS hoa_don (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    ma_hoa_don VARCHAR(40) NOT NULL UNIQUE,
    khach_hang_id BIGINT,
    ten_khach_hang VARCHAR(150) NOT NULL,
    tong_tien DECIMAL(15,2) NOT NULL,
    phuong_thuc_thanh_toan VARCHAR(40) NOT NULL,
    trang_thai VARCHAR(30) NOT NULL DEFAULT 'DA_THANH_TOAN',
    ngay_tao DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_hoa_don_khach_hang FOREIGN KEY (khach_hang_id)
        REFERENCES khach_hang(id) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS chi_tiet_hoa_don (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    hoa_don_id BIGINT NOT NULL,
    san_pham_id BIGINT,
    ten_san_pham VARCHAR(180) NOT NULL,
    ma_san_pham VARCHAR(40) NOT NULL,
    don_gia DECIMAL(15,2) NOT NULL,
    so_luong INT NOT NULL,
    thanh_tien DECIMAL(15,2) NOT NULL,
    bao_hanh_den DATE,
    CONSTRAINT chk_chi_tiet_so_luong CHECK (so_luong > 0),
    CONSTRAINT fk_chi_tiet_hoa_don FOREIGN KEY (hoa_don_id)
        REFERENCES hoa_don(id) ON DELETE CASCADE,
    CONSTRAINT fk_chi_tiet_san_pham FOREIGN KEY (san_pham_id)
        REFERENCES san_pham(id) ON DELETE SET NULL
);

SET @mobihub_ddl = IF((SELECT COUNT(*) FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_SCHEMA=DATABASE() AND TABLE_NAME='chi_tiet_hoa_don' AND COLUMN_NAME='bao_hanh_den')=0,
    'ALTER TABLE chi_tiet_hoa_don ADD COLUMN bao_hanh_den DATE NULL', 'SELECT 1');
PREPARE mobihub_stmt FROM @mobihub_ddl;
EXECUTE mobihub_stmt;
DEALLOCATE PREPARE mobihub_stmt;

UPDATE chi_tiet_hoa_don ci
JOIN hoa_don hd ON hd.id=ci.hoa_don_id
JOIN san_pham sp ON sp.id=ci.san_pham_id
SET ci.bao_hanh_den=DATE_ADD(DATE(hd.ngay_tao), INTERVAL sp.thoi_han_bao_hanh_thang MONTH)
WHERE ci.bao_hanh_den IS NULL;

CREATE TABLE IF NOT EXISTS thong_tin_giao_hang (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    hoa_don_id BIGINT NOT NULL UNIQUE,
    dia_chi_giao_hang VARCHAR(500) NOT NULL,
    CONSTRAINT fk_giao_hang_hoa_don FOREIGN KEY (hoa_don_id)
        REFERENCES hoa_don(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS danh_gia (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    san_pham_id BIGINT NOT NULL,
    khach_hang_id BIGINT NOT NULL,
    hoa_don_id BIGINT NOT NULL,
    so_sao INT NOT NULL,
    noi_dung VARCHAR(1500),
    trang_thai VARCHAR(20) NOT NULL DEFAULT 'CHO_DUYET',
    ngay_tao DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_danh_gia_so_sao CHECK (so_sao BETWEEN 1 AND 5),
    CONSTRAINT uk_danh_gia_khach_hoa_don_sp UNIQUE (khach_hang_id, hoa_don_id, san_pham_id),
    CONSTRAINT fk_danh_gia_san_pham FOREIGN KEY (san_pham_id)
        REFERENCES san_pham(id) ON DELETE CASCADE,
    CONSTRAINT fk_danh_gia_khach_hang FOREIGN KEY (khach_hang_id)
        REFERENCES khach_hang(id) ON DELETE CASCADE,
    CONSTRAINT fk_danh_gia_hoa_don FOREIGN KEY (hoa_don_id)
        REFERENCES hoa_don(id) ON DELETE CASCADE
);

ALTER TABLE danh_gia MODIFY so_sao INT NOT NULL;
